import numpy as np
from typing import List

def priority(item: float, bins_remain_cap: np.ndarray) -> np.ndarray:
    """
    Priority heuristic that uses quantile-driven common-size estimation to adaptively 
    select target residual capacities based on the relative position of the current 
    item within the spectrum of available bin remaining capacities.
    
    The approach computes robust statistical proxies (median and interquartile range) 
    of bins_remain_cap to classify the current item as 'large', 'medium', or 'small'. 
    This classification determines the target residual capacities to prioritize, 
    leveraging the observed packing context without maintaining external state.
    
    Implementation idea: Instead of using only multiples of the current item as in 
    parent solutions, this approach analyzes the distribution of remaining bin 
    capacities to infer likely future item sizes and adapt the priority scoring 
    accordingly. For example, if the current item is relatively small compared to 
    available bin spaces, we might expect more small items and thus prioritize 
    leaving fractional capacities like 0.25 or 0.5 of full capacity.
    
    Args:
        item: Size of the item to place
        bins_remain_cap: NumPy array of remaining bin capacities
        
    Returns:
        NumPy array of priority scores for each bin
    """
    # Calculate post-placement remaining capacities
    post_placement_caps = bins_remain_cap - item
    
    # Initialize scores
    scores = np.full_like(bins_remain_cap, -np.inf, dtype=float)
    
    # Only consider bins that can accommodate the item
    feasible_bins = bins_remain_cap >= item
    
    if not np.any(feasible_bins):
        return scores  # All bins remain with -inf scores
    
    feasible_post_caps = post_placement_caps[feasible_bins]
    feasible_scores = np.zeros_like(feasible_post_caps, dtype=float)
    
    # Compute quantiles of the remaining bin capacities to understand the context
    if len(bins_remain_cap) > 0:
        q25 = np.percentile(bins_remain_cap, 25)
        q50 = np.percentile(bins_remain_cap, 50)  # median
        q75 = np.percentile(bins_remain_cap, 75)
    else:
        # Fallback if no bins are available
        q25 = q50 = q75 = 1.0
    
    # Classify the current item based on its size relative to bin capacity distribution
    if item <= q25:
        # Small item relative to remaining capacities - likely more small items coming
        # Target residuals that are useful for small items (fractional bin capacities)
        target_residuals = [q50, q25, q75, 0.25, 0.5, 0.75]  # Use both quantiles and common fractions
    elif item <= q50:
        # Medium item - balanced approach
        target_residuals = [q50, q25, q75, item, item * 0.5, item * 1.5]
    else:
        # Large item - focus on best fit to avoid fragmentation
        target_residuals = [item, item * 0.5, item * 1.5, q50]
    
    # Remove non-positive targets
    target_residuals = [target for target in target_residuals if target > 0]
    
    # Calculate how close the remaining capacity is to target residuals
    multiple_fit_score = np.zeros_like(feasible_post_caps, dtype=float)
    
    for target in target_residuals:
        # Calculate distance to target
        distances = np.abs(feasible_post_caps - target)
        # Add to the score (higher score for closer matches)
        multiple_fit_score += 1.0 / (1.0 + distances)
    
    # Add Best Fit component: prefer bins with less remaining space after placement
    # This prevents overfilling bins unnecessarily
    best_fit_component = -feasible_post_caps  # Higher score for less remaining space after placement
    
    # Combine both components with appropriate weights
    # Using a smaller weight for best fit component as in successful parent solutions
    feasible_scores = multiple_fit_score + 0.05 * best_fit_component
    
    # Assign the calculated scores to the correct positions
    scores[feasible_bins] = feasible_scores
    
    return scores

import numpy as np
from typing import List

def priority(item: float, bins_remain_cap: np.ndarray) -> np.ndarray:
    """
    Priority heuristic that assigns higher priority to bins that, after placement, 
    result in a remaining capacity that is close to multiples of common item sizes.
    This implementation estimates common item sizes based on recent items and 
    prioritizes bins that leave "useful" remaining capacities.
    
    The approach combines:
    1. Best Fit principle (prioritizing bins with less remaining space)
    2. Multiple-based scoring (favoring bins that leave capacities near common sizes)
    
    Args:
        item: Size of the item to place
        bins_remain_cap: NumPy array of remaining bin capacities
        
    Returns:
        NumPy array of priority scores for each bin
    """
    # Calculate post-placement remaining capacities
    post_placement_caps = bins_remain_cap - item
    
    # Initialize scores
    scores = np.full_like(bins_remain_cap, -np.inf, dtype=float)
    
    # Only consider bins that can accommodate the item
    feasible_bins = bins_remain_cap >= item
    
    if not np.any(feasible_bins):
        return scores  # All bins remain with -inf scores
    
    feasible_post_caps = post_placement_caps[feasible_bins]
    feasible_scores = np.zeros_like(feasible_post_caps, dtype=float)
    
    # Estimate common item sizes based on the item being placed
    current_item = item
    estimated_common_sizes = [current_item, current_item * 0.9, current_item * 1.1, 
                              current_item * 0.5, current_item * 1.5, current_item * 0.75, current_item * 1.25]
    
    # Filter out non-positive sizes
    estimated_common_sizes = [size for size in estimated_common_sizes if size > 0]
    
    # Calculate how close the remaining capacity is to being a multiple of common_sizes
    multiple_fit_score = np.zeros_like(feasible_post_caps, dtype=float)
    
    for common_size in estimated_common_sizes:
        # Calculate distances to the nearest multiple of common_size
        multipliers = feasible_post_caps / common_size
        lower_mult = np.floor(multipliers)
        upper_mult = lower_mult + 1
        
        dist_to_lower = np.abs(feasible_post_caps - lower_mult * common_size)
        dist_to_upper = np.abs(feasible_post_caps - upper_mult * common_size)
        
        min_distances = np.minimum(dist_to_lower, dist_to_upper)
        
        # Add to the score (higher score for closer matches)
        multiple_fit_score += 1.0 / (1.0 + min_distances)
    
    # Add Best Fit component: prefer bins with less remaining space
    # This prevents overfilling bins unnecessarily
    best_fit_component = -feasible_post_caps  # Higher score for less remaining space
    
    # Combine both components
    # Using smaller weight for best fit component to balance between multiple-based scoring and best fit
    # Based on parent solution analysis, 0.05 worked well, but let's try slightly different values
    feasible_scores = multiple_fit_score + 0.04 * best_fit_component
    
    # Assign the calculated scores to the correct positions
    scores[feasible_bins] = feasible_scores
    
    return scores
