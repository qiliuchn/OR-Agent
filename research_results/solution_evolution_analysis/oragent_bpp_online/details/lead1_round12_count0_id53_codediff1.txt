import numpy as np
from typing import List

def priority(item: float, bins_remain_cap: np.ndarray) -> np.ndarray:
    """
    Priority heuristic using non-linear, distance-weighted scoring for common-size matching.
    Instead of linear inverse-distance weighting (1/(1+distance)), this implementation uses
    exponential decay to strongly favor exact matches while gently penalizing moderate gaps,
    but sharply discounting large mismatches. This creates a more discriminative scoring
    function that better prioritizes high-utility residual spaces.
    
    Implementation idea: Replace the current linear inverse-distance weighting with a
    non-linear function such as exponential decay (exp(-distance/scale)) or sigmoidal
    penalties that strongly differentiate between exact matches and near-matches while
    heavily penalizing poor fits. This creates sharper distinctions in priority scores,
    leading to more decisive bin selection that favors strategic space allocation over
    suboptimal placements.

    Args:
        item: Size of the item to place
        bins_remain_cap: NumPy array of remaining bin capacities
        
    Returns:
        NumPy array of priority scores for each bin
    """
    # Calculate post-placement remaining capacities
    post_placement_caps = bins_remain_cap - item
    
    # Initialize scores
    scores = np.full_like(bins_remain_cap, -np.inf, dtype=float)
    
    # Only consider bins that can accommodate the item
    feasible_bins = bins_remain_cap >= item
    
    if not np.any(feasible_bins):
        return scores  # All bins remain with -inf scores
    
    feasible_post_caps = post_placement_caps[feasible_bins]
    feasible_scores = np.zeros_like(feasible_post_caps, dtype=float)
    
    # Define estimated common sizes based on the current item
    estimated_common_sizes = [
        item,
        item * 0.5,
        item * 1.5,
        item * 0.75,
        item * 1.25,
        item * 0.33,
        item * 0.67,
        item * 0.25,
        item * 0.125,  # Finer granularity
        item * 0.875,
        item * 0.167,  # 1/6 fraction
        item * 0.833,  # 5/6 fraction
        item * 0.2,    # 1/5 fraction
        item * 0.4,    # 2/5 fraction
        item * 0.6,    # 3/5 fraction
        item * 0.8,    # 4/5 fraction
        item * 0.1,    # Even finer granularity
        item * 0.9,    # 9/10 fraction
        item * 0.0625, # 1/16 fraction for extremely fine granularity
        item * 0.375   # 3/8 fraction
    ]
    
    # Filter out non-positive sizes
    estimated_common_sizes = [size for size in estimated_common_sizes if size > 0]
    
    # Calculate how close the remaining capacity is to being a multiple of common sizes
    # Using exponential decay for non-linear scoring
    multiple_fit_score = np.zeros_like(feasible_post_caps, dtype=float)
    
    for common_size in estimated_common_sizes:
        # Calculate how many multiples of common_size fit in the remaining capacity
        max_multiplier = int(np.max(feasible_post_caps) // common_size) + 2
        if max_multiplier > 0:
            # Calculate distances to all possible multiples
            distances_to_multiples = np.min([
                np.abs(feasible_post_caps - n * common_size) 
                for n in range(0, max_multiplier)
            ], axis=0)

            # A reasonable scale might be related to the common size itself
            scale = max(common_size * 0.1, 0.1)  # Prevent too-small scales
            # Add to the score (higher score for closer matches)
            multiple_fit_score += np.exp(-distances_to_multiples / scale)
    
    # Add Best Fit component: prefer bins with less remaining space
    # This prevents overfilling bins unnecessarily
    # Using a gentle penalty to maintain balance with common-size matching
    best_fit_weight = 0.05
    best_fit_component = -feasible_post_caps  # Higher score for less remaining space
    
    # Combine both components
    feasible_scores = multiple_fit_score + best_fit_weight * best_fit_component
    
    # Assign the calculated scores to the correct positions
    scores[feasible_bins] = feasible_scores
    
    return scores

    activates fine-grained fractional multipliers and diversity-aware scoring. This refines 
    previous proxy-based inference with more robust statistical signals while remaining fully 
    stateless.
    
    Implementation idea: Compute what percentile the current item falls into when compared 
    against the distribution of remaining bin capacities. Use this quantile information to 
    determine if the item is unusually large/small relative to current context, and select 
    appropriate multiplier sets and scoring strategies. This creates more nuanced adaptive
