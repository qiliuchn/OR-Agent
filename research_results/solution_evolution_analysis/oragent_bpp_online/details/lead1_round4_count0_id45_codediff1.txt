import numpy as np

def priority(item: float, bins_remain_cap: np.ndarray) -> np.ndarray:
    """
    Dynamic priority function that adaptively selects between Best Fit and Worst Fit heuristics
    based on the current item size relative to an adaptive threshold. The threshold is updated
    using an exponential moving average of recent item sizes to adapt to distribution shifts.
    
    The intuition is that large items (> threshold * bin_capacity) should use Best Fit to pack
    tightly and avoid fragmentation, while small items (<= threshold * bin_capacity) should use
    Worst Fit to preserve space for larger items that may come later.
    
    This builds on the success of Best Fit with controlled randomness from parent solutions,
    but adds adaptive strategy selection based on item size characteristics.
    
    Args:
        item: Size of the item to be placed
        bins_remain_cap: Array of remaining capacities for available bins
        
    Returns:
        Array of priority scores for each bin (higher score means higher priority)
    """
    # Initialize class-level variables to maintain state across calls
    if not hasattr(priority, 'ema_item_size'):
        # Initialize EMA with a reasonable guess (half of typical bin capacity)
        priority.ema_item_size = 50.0  # Assuming typical bin capacity around 100
    if not hasattr(priority, 'alpha'):
        # EMA smoothing factor - balance between responsiveness and stability
        priority.alpha = 0.05  # Slow adaptation to prevent overfitting to recent items
    
    # Update the exponential moving average of item sizes
    priority.ema_item_size = priority.alpha * item + (1 - priority.alpha) * priority.ema_item_size
    
    # Determine adaptive threshold based on EMA of item sizes
    # Use a heuristic that considers both the EMA and the typical bin capacity
    bin_capacity_estimate = np.max(bins_remain_cap) if len(bins_remain_cap) > 0 else 100.0
    threshold = min(0.5, max(0.2, priority.ema_item_size / bin_capacity_estimate))
    
    # Determine feasibility
    feasible = bins_remain_cap >= item
    
    # Select heuristic based on item size relative to threshold
    if item > threshold * bin_capacity_estimate:
        # Large item: Use Best Fit (pack tightly)
        # Higher priority for bins with least remaining capacity (after placement)
        scores = np.where(feasible, -bins_remain_cap, -np.inf)
    else:
        # Small item: Use Worst Fit (preserve space for larger items)
        # Higher priority for bins with most remaining capacity
        scores = np.where(feasible, bins_remain_cap, -np.inf)
    
    # Add controlled randomness similar to parent solutions to escape local optima
    # Only add randomness when multiple feasible bins exist
    if len(bins_remain_cap) > 1 and np.sum(feasible) > 1:
        # Calculate a dynamic noise scale based on bin capacity
        noise_scale = 0.005 * bin_capacity_estimate
        random_noise = np.random.uniform(-noise_scale, noise_scale, size=bins_remain_cap.shape)
        scores = scores + np.where(feasible, random_noise, 0.0)
    
    return scores

import numpy as np

def priority(item: float, bins_remain_cap: np.ndarray) -> np.ndarray:
    """
    Adaptive priority function based on parent solution #2 with improved randomness control.
    Uses Best Fit as the base heuristic with controlled randomness to escape local optima,
    but with a more sophisticated approach to determine when to add randomness based on
    packing density and bin utilization.
    
    This approach combines the effectiveness of Best Fit heuristic with controlled randomness 
    to explore alternative packing configurations when beneficial, potentially leading to 
    better global packing efficiency.
    
    Args:
        item: Size of the item to be placed
        bins_remain_cap: Array of remaining capacities for available bins
        
    Returns:
        Array of priority scores for each bin (higher score means higher priority)
    """
    # Calculate initial statistics
    initial_capacity = np.max(bins_remain_cap) if len(bins_remain_cap) > 0 else item
    active_bins_count = len(bins_remain_cap)  # Number of bins that can fit the current item
    
    # Calculate packing density: how full the available bins are on average
    if len(bins_remain_cap) > 0:
        avg_remaining_ratio = np.mean(bins_remain_cap) / initial_capacity
        packing_density = 1.0 - avg_remaining_ratio  # How much of the capacity is used
    else:
        packing_density = 0.0
    
    scores = np.zeros_like(bins_remain_cap, dtype=float)
    
    # Determine feasibility
    feasible = bins_remain_cap >= item
    
    # Apply Best Fit as base heuristic (higher priority to bins with less remaining space before placement)
    base_scores = np.where(feasible, -bins_remain_cap, -np.inf)
    
    # Determine when to add randomness based on how many bins are in play
    # Following parent solution #2 approach but with slight refinement
    total_bins_estimate = len(bins_remain_cap) + int(initial_capacity * 0.5)  # Rough estimate of total bins initially created
    randomness_threshold = 0.3  # Start adding randomness when 30% of bins are active
    
    should_add_randomness = active_bins_count / total_bins_estimate > randomness_threshold and len(bins_remain_cap) > 1
    
    if should_add_randomness:
        # Add small random perturbation to break ties and explore alternatives
        noise_scale = 0.01 * initial_capacity  # Small noise relative to bin capacity (matching parent #2)
        random_noise = np.random.uniform(-noise_scale, noise_scale, size=bins_remain_cap.shape)
        scores = base_scores + np.where(feasible, random_noise, 0.0)
    else:
        # Use pure Best Fit heuristic when few bins are active
        scores = base_scores
    
    return scores
