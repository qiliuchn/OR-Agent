def priority(item: float, bins_remain_cap: np.ndarray) -> np.ndarray:
    """
    Adaptive priority function that maintains a running histogram of item sizes
    and uses it to score bins based on how well their post-placement remaining
    capacity matches high-frequency item sizes.
    
    The approach:
    1. Maintains a global history of item sizes to build a histogram
    2. Uses this histogram to identify high-frequency item sizes
    3. Prioritizes bins whose post-placement capacity aligns with these frequent sizes
    4. Combines this with best-fit principles to avoid wasting space
    
    Implementation considerations:
    - Uses exponential moving average to maintain item size frequencies
    - Updates the histogram each time this function is called
    - Balances between matching frequent sizes and following best-fit principle
    - Handles edge cases where no bins can accommodate the item
    
    Args:
        item: Size of the item to place
        bins_remain_cap: NumPy array of remaining bin capacities
        
    Returns:
        NumPy array of priority scores for each bin
    """
    # Use module-level variables to maintain state across calls
    if not hasattr(priority, 'item_histogram'):
        # Initialize histogram parameters
        priority.bin_edges = np.linspace(0, 100, 21)  # Assuming max item size around 100
        priority.bin_centers = (priority.bin_edges[:-1] + priority.bin_edges[1:]) / 2
        priority.item_counts = np.zeros(len(priority.bin_centers))  # Exponential moving average weights
        priority.total_items = 0
        priority.decay_factor = 0.995  # How quickly old items are forgotten
        
    # Update the histogram with the current item
    priority.total_items += 1
    # Decay previous counts
    priority.item_counts *= priority.decay_factor
    # Find which bin the current item belongs to
    item_bin_idx = np.digitize(item, priority.bin_edges) - 1
    # Make sure index is within bounds
    item_bin_idx = np.clip(item_bin_idx, 0, len(priority.bin_centers) - 1)
    # Increment count for this bin
    priority.item_counts[item_bin_idx] += 1

    threshold = np.percentile(priority.item_counts, 75)  # Top 25% of frequencies
    frequent_sizes = priority.bin_centers[priority.item_counts >= threshold]

    frequent_sizes = frequent_sizes[frequent_sizes > 0]

    if len(frequent_sizes) == 0:

        frequent_sizes = [size for size in frequent_sizes if size > 0]

    post_placement_caps = bins_remain_cap - item
    
    # Initialize scores
    scores = np.full_like(bins_remain_cap, -np.inf, dtype=float)
    
    # Only consider bins that can accommodate the item
    feasible_bins = bins_remain_cap >= item
    
    if not np.any(feasible_bins):
        return scores  # All bins remain with -inf scores

    feasible_post_caps = post_placement_caps[feasible_bins]

        distances = np.abs(feasible_post_caps - freq_size)

        feasible_scores += 1.0 / (1.0 + distances)

        else:
            best_fit_component = np.zeros_like(post_placement_caps)
    
    # Combine both components with balanced weights
    combined_scores = feasible_scores + 0.2 * best_fit_component  # Weight best fit moderately
    
    # Assign the calculated scores to the correct positions
    scores[feasible_bins] = combined_scores
    
    return scores

    1. Estimation of common item sizes based on the current item

    3. Best Fit principle as a secondary criterion to avoid wasting space

        item: Size of the item to place

    Returns:
