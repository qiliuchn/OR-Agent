def priority(item: float, bins_remain_cap: np.ndarray) -> np.ndarray:
    """
    Priority heuristic that combines a comprehensive multiplier set with a lightweight
    stateful component that tracks recent item sizes in a sliding window to dynamically
    adjust multiplier weights based on observed empirical frequencies. This addresses
    the limitation of purely stateless heuristics by allowing adaptation to local
    distribution shifts while maintaining computational efficiency.
    
    The approach combines:
    1. Estimation of common item sizes based on the current item and comprehensive multiplier set
    2. Lightweight state tracking of recent items in a sliding window to adapt multiplier weights
    3. Scoring bins based on how well their post-placement capacity matches adapted estimates
    4. Best Fit principle as a secondary criterion to avoid wasting space
    
    Args:
        item: Size of the item to place
        bins_remain_cap: NumPy array of remaining bin capacities
        
    Returns:
        NumPy array of priority scores for each bin
    """

    if not hasattr(priority, 'recent_items'):
        priority.recent_items = []

    priority.recent_items.append(item)
    if len(priority.recent_items) > 10:
        priority.recent_items.pop(0)

    adaptive_multipliers = []
    adaptive_weights = []

    base_multipliers = [1.0, 0.5, 0.75, 0.9, 1.1, 1.25, 1.5, 0.25, 2.0, 0.33, 0.66, 1.33, 0.2, 0.4, 0.6, 0.8, 1.2, 1.4, 1.6, 1.8]

    if len(priority.recent_items) >= 3:
        # Analyze recent items to find common ratios
        recent_ratios = []
        for i in range(len(priority.recent_items)):
            for j in range(i+1, len(priority.recent_items)):
                if priority.recent_items[j] != 0:
                    ratio = priority.recent_items[i] / priority.recent_items[j]
                    recent_ratios.extend([ratio, 1/ratio])

        recent_ratios = [r for r in recent_ratios if 0.1 <= r <= 3.0]

        unique_ratios = list(set(recent_ratios))
        for ratio in unique_ratios:

            closest_base = min(base_multipliers, key=lambda x: abs(x - ratio))

                continue

                adaptive_multipliers.append(ratio)
                # Weight based on frequency in recent items
