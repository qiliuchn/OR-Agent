import numpy as np
from typing import List

def priority(item: float, bins_remain_cap: np.ndarray) -> np.ndarray:
    """
    Priority heuristic implementing adaptive weight modulation based on packing progress proxies.
    
    This implementation builds upon the successful dynamic common-size generation approach from
    the parent solution, but adds adaptive weight modulation based on within-call metrics
    that capture the current state of packing progress. Specifically, it computes:
    1. Fraction of bins with very low remaining capacity (< 0.1 of original capacity)
    2. Dispersion measure of feasible remainders
    These metrics are used to dynamically adjust the balance between best-fit and multiple-fit
    components as the packing evolves, without requiring cross-call state.
    
    The core insight is that early in packing (when most bins have high capacity), multiple-fit
    strategies work better to create common sizes for future items. Later in packing (when many
    bins have low remaining capacity), best-fit becomes more important to avoid creating new bins.
    
    Args:
        item: Size of the item to place
        bins_remain_cap: NumPy array of remaining bin capacities
        
    Returns:
        NumPy array of priority scores for each bin
    """
    # Calculate post-placement remaining capacities
    post_placement_caps = bins_remain_cap - item
    
    # Initialize scores
    scores = np.full_like(bins_remain_cap, -np.inf, dtype=float)
    
    # Only consider bins that can accommodate the item
    feasible_bins = bins_remain_cap >= item
    
    if not np.any(feasible_bins):
        return scores  # All bins remain with -inf scores
    
    feasible_post_caps = post_placement_caps[feasible_bins]
    feasible_scores = np.zeros_like(feasible_post_caps, dtype=float)
    
    # Calculate the percentile rank of the current item among remaining bin capacities
    if len(bins_remain_cap) > 0:
        # Calculate the percentile of the current item in the context of remaining capacities
        sorted_caps = np.sort(bins_remain_cap)
        item_percentile = np.searchsorted(sorted_caps, item) / len(sorted_caps)
    else:
        item_percentile = 0.5  # Default if no bins exist
    
    # Compute packing progress proxies
    # 1. Fraction of bins with very low remaining capacity (threshold = 0.1 * original capacity)
    # Assuming original capacity can be estimated from max current capacity or typical value
    if len(bins_remain_cap) > 0:
        # Estimate original capacity as max remaining capacity or use a reasonable proxy
        est_capacity = np.max(bins_remain_cap) if len(bins_remain_cap) > 0 else 100.0
        low_capacity_threshold = 0.1 * est_capacity
        low_capacity_fraction = np.sum(bins_remain_cap < low_capacity_threshold) / len(bins_remain_cap)
    else:
        low_capacity_fraction = 0.0
    
    # 2. Dispersion of feasible remainders (coefficient of variation)
    if len(feasible_post_caps) > 1:
        mean_remainder = np.mean(feasible_post_caps)
        std_remainder = np.std(feasible_post_caps)
        if mean_remainder > 1e-9:  # Avoid division by zero
            remainder_dispersion = std_remainder / mean_remainder
        else:
            remainder_dispersion = 0.0
    else:
        remainder_dispersion = 0.0
    
    # Modulate base weights based on packing progress
    # As low_capacity_fraction increases, favor best-fit over multiple-fit
    # As remainder_dispersion decreases (remainders become more similar), favor multiple-fit
    progress_modulation_factor = 1.0 - low_capacity_fraction  # Higher = more multiple-fit focus
    dispersion_modulation_factor = 1.0 + remainder_dispersion  # Higher = more multiple-fit focus
    
    # Define category-specific base weights based on refined percentile thresholds
    if item_percentile > 0.82:  # Very large item
        base_best_fit_weight, base_multiple_fit_weight = 0.25 * 1.07, 0.75 * 0.99  # Enhanced best-fit for large items
    elif item_percentile < 0.18:  # Very small item
        base_best_fit_weight, base_multiple_fit_weight = 0.001 * 0.93, 1.194 * 1.01  # Reduced best-fit for small items
    elif item_percentile > 0.62:  # Large-medium item
        base_best_fit_weight, base_multiple_fit_weight = 0.14, 0.86  # Between medium and large
    elif item_percentile < 0.38:  # Small-medium item
        base_best_fit_weight, base_multiple_fit_weight = 0.025, 1.055  # Between medium and small
    else:  # True medium item
        base_best_fit_weight, base_multiple_fit_weight = 0.08, 0.92  # Medium weights
    
    # Apply progress-based modulation to base weights
    modulated_best_fit_weight = base_best_fit_weight * (2.0 - progress_modulation_factor)
    modulated_multiple_fit_weight = base_multiple_fit_weight * progress_modulation_factor * dispersion_modulation_factor
    
    # Normalize weights to preserve overall magnitude
    total_base_weight = base_best_fit_weight + base_multiple_fit_weight
    total_modulated_weight = modulated_best_fit_weight + modulated_multiple_fit_weight
    
    if total_modulated_weight > 0 and total_base_weight > 0:
        scaling_factor = total_base_weight / total_modulated_weight
        modulated_best_fit_weight *= scaling_factor
        modulated_multiple_fit_weight *= scaling_factor
    
    # Generate dynamic common sizes using context-aware sampling
    if len(feasible_post_caps) > 0:
        # Identify important scales from the current remainder distribution
        unique_caps = np.unique(feasible_post_caps)
        if len(unique_caps) > 0:
            # Sample key values from the current distribution as common sizes
            # Quantiles provide good coverage of the distribution
            quantiles = np.array([0.1, 0.25, 0.33, 0.5, 0.67, 0.75, 0.9])
            quantile_values = np.quantile(unique_caps, quantiles)
            
            # Also include some ratios of the current item size
            item_ratios = np.array([item, item * 0.5, item * 1.5, item * 0.75, item * 1.25])
            
            # Include some simple fractions of the current remainders
            remainder_fractions = np.concatenate([
                unique_caps * 0.5,
                unique_caps * 0.33,
                unique_caps * 0.67,
                unique_caps * 0.25,
                unique_caps * 0.75
            ])
            
            # Combine all dynamically generated common sizes
            dynamic_common_sizes = np.concatenate([
                quantile_values,
                item_ratios,
                remainder_fractions
            ])
            
            # Remove duplicates and very small values
            dynamic_common_sizes = np.unique(dynamic_common_sizes)
            dynamic_common_sizes = dynamic_common_sizes[dynamic_common_sizes > 1e-9]
        else:
            # Fallback if no unique values exist
            dynamic_common_sizes = np.array([item, item * 0.5, item * 1.5])
    else:
        dynamic_common_sizes = np.array([item, item * 0.5, item * 1.5])
    
    # Calculate multiple fit score based on dynamic common sizes
    multiple_fit_score = np.zeros_like(feasible_post_caps, dtype=float)
    
    # Limit the number of common sizes to prevent excessive computation
    max_common_sizes = 50
    if len(dynamic_common_sizes) > max_common_sizes:
        # Take the first max_common_sizes values to maintain consistency
        dynamic_common_sizes = dynamic_common_sizes[:max_common_sizes]
    
    for common_size in dynamic_common_sizes:
        if common_size > 1e-9:  # Avoid division by very small numbers
            # Calculate distance to nearest multiple of the common size
            # Consider multiples up to a reasonable limit
            max_multiplier = min(int(np.max(feasible_post_caps) // common_size) + 2, 10)
            if max_multiplier > 0:
                # Calculate distances to all possible multiples
                distances_to_multiples = np.min([
                    np.abs(feasible_post_caps - n * common_size) 
                    for n in range(0, max_multiplier)
                ], axis=0)
                
                # Add to the score (higher score for closer matches)
                multiple_fit_score += 1.0 / (1.0 + distances_to_multiples)
    
    # Add Best Fit component: prefer bins with less remaining space
    best_fit_component = -feasible_post_caps  # Higher score for less remaining space
    
    # Combine all components with modulated weights
    feasible_scores = (
        modulated_multiple_fit_weight * multiple_fit_score + 
        modulated_best_fit_weight * best_fit_component
    )
    
    # Assign the calculated scores to the correct positions
    scores[feasible_bins] = feasible_scores
    
    return scores
