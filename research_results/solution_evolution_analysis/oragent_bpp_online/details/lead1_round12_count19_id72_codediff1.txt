import numpy as np
from typing import List

def priority(item: float, bins_remain_cap: np.ndarray) -> np.ndarray:
    """
    Multi-objective priority heuristic that balances three components: (1) best-fit (minimizing 
    immediate waste), (2) multiple-fit alignment (preserving structured residual space), and 
    (3) bin diversity preservation (avoiding premature fragmentation by penalizing placements 
    that reduce the variety of remaining capacities). The relative importance of each objective 
    is dynamically adjusted based on quantile-based item size classification and distributional 
    features like entropy or Gini coefficient of bin remainders. This extends the current 
    two-component scoring to address the observed ~25-bin gap to optimality by better managing 
    long-term packing flexibility.
    
    Implementation builds upon the successful quantile-adaptive approach from parent solutions,
    adding a third diversity preservation component that penalizes placements reducing capacity
    variety, and adjusts weights dynamically based on entropy of remaining capacities.

    Args:
        item: Size of the item to place
        bins_remain_cap: NumPy array of remaining bin capacities
        
    Returns:
        NumPy array of priority scores for each bin
    """
    # Calculate post-placement remaining capacities
    post_placement_caps = bins_remain_cap - item
    
    # Initialize scores
    scores = np.full_like(bins_remain_cap, -np.inf, dtype=float)
    
    # Only consider bins that can accommodate the item
    feasible_bins = bins_remain_cap >= item
    
    if not np.any(feasible_bins):
        return scores  # All bins remain with -inf scores
    
    feasible_post_caps = post_placement_caps[feasible_bins]
    feasible_scores = np.zeros_like(feasible_post_caps, dtype=float)
    
    # Calculate the percentile rank of the current item among remaining bin capacities
    if len(bins_remain_cap) > 0:
        # Calculate the percentile of the current item in the context of remaining capacities
        sorted_caps = np.sort(bins_remain_cap[bins_remain_cap > 0])  # Only consider non-empty bins
        if len(sorted_caps) > 0:
            # Find where the item would fit in the sorted array
            item_percentile = np.searchsorted(sorted_caps, item) / len(sorted_caps)
        else:
            item_percentile = 0.5  # Default if no bins have capacity
    else:
        item_percentile = 0.5  # Default if no bins exist
    
    # Calculate entropy of remaining capacities to measure diversity
    positive_caps = bins_remain_cap[bins_remain_cap > 0]
    if len(positive_caps) > 1:
        # Normalize capacities to form a probability distribution
        total_capacity = np.sum(positive_caps)
        if total_capacity > 0:
            probs = positive_caps / total_capacity
            # Calculate Shannon entropy
            entropy = -np.sum(probs * np.log2(probs + 1e-10))  # Add small value to avoid log(0)
            # Normalize entropy by max possible entropy (log2 of number of unique bins)
            max_entropy = np.log2(len(positive_caps)) if len(positive_caps) > 0 else 1
            normalized_entropy = entropy / max_entropy if max_entropy > 0 else 0
        else:
            normalized_entropy = 0
    else:
        normalized_entropy = 0
    
    # Define learned multiplier embeddings based on size category - reduced size to improve performance
    learned_embeddings = {
        'large': [
            item,
            item * 0.5,
            item * 1.5,
            item * 0.75,
            item * 1.25
        ],
        'small': [
            item,
            item * 0.5,
            item * 0.25,
            item * 0.75,
            item * 0.33,
            item * 0.67
        ],
        'medium': [
            item,
            item * 0.5,
            item * 1.5,
            item * 0.75,
            item * 1.25,
            item * 0.33,
            item * 0.67,
            item * 0.25
        ]
    }
    
    # Define weights for the three components: (best_fit, multiple_fit, diversity_preservation)
    learned_weights = {
        'large': (0.23, 0.77, 0.1),   # Slightly emphasize diversity for large items
        'small': (0.001, 1.19, 0.05),  # Less emphasis on diversity for small items
        'medium': (0.065, 0.935, 0.08)  # Balanced with some diversity consideration
    }
    
    # Determine the size category based on percentile with refined thresholds
    if item_percentile > 0.80:  # Very large item
        estimated_common_sizes = learned_embeddings['large']
        best_fit_weight, multiple_fit_weight, diversity_weight = learned_weights['large']
        # Slightly increase best-fit weight for very large items to ensure they fit
        best_fit_weight *= 1.05
    elif item_percentile < 0.20:  # Very small item
        estimated_common_sizes = learned_embeddings['small']
        best_fit_weight, multiple_fit_weight, diversity_weight = learned_weights['small']
        # Further reduce best-fit weight for very small items
        best_fit_weight *= 0.95
    elif item_percentile > 0.60:  # Large-medium item
        # Interpolate between medium and large strategies
        estimated_common_sizes = learned_embeddings['medium']
        # Adjust weights to be between medium and large values
        best_fit_weight, multiple_fit_weight, diversity_weight = 0.13, 0.87, 0.07
    elif item_percentile < 0.40:  # Small-medium item
        # Interpolate between medium and small strategies
        estimated_common_sizes = learned_embeddings['small']  # Use small's rich multiplier set
        # Adjust weights to be between medium and small values
        best_fit_weight, multiple_fit_weight, diversity_weight = 0.03, 1.05, 0.06
    else:  # True medium item
        estimated_common_sizes = learned_embeddings['medium']
        best_fit_weight, multiple_fit_weight, diversity_weight = learned_weights['medium']
    
    # Adjust weights based on entropy of remaining capacities
    # When entropy is low (bins are similar), emphasize diversity preservation more
    entropy_factor = 1.0 - normalized_entropy  # Higher factor when entropy is low
    diversity_weight *= (0.8 + 0.4 * entropy_factor)  # Scale diversity weight based on entropy
    
    # Filter out non-positive sizes
    estimated_common_sizes = [size for size in estimated_common_sizes if size > 0]
    
    # Calculate how close the remaining capacity is to being a multiple of common sizes
    multiple_fit_score = np.zeros_like(feasible_post_caps, dtype=float)
    
    # Optimized calculation to avoid creating large intermediate arrays
    for common_size in estimated_common_sizes:
        # Calculate how many multiples of common_size fit in the remaining capacity
        max_multiplier = int(np.max(feasible_post_caps) // common_size) + 2
        if max_multiplier > 0:
            # Calculate distances to all possible multiples efficiently
            min_distances = np.full_like(feasible_post_caps, np.inf, dtype=float)
            for n in range(0, min(max_multiplier, 10)):  # Limit to avoid excessive computation
                distances = np.abs(feasible_post_caps - n * common_size)
                min_distances = np.minimum(min_distances, distances)
            
            # Add to the score (higher score for closer matches)
            # Use a small epsilon to avoid division by zero
            multiple_fit_score += 1.0 / (1.0 + min_distances)
    
    # Add Best Fit component: prefer bins with less remaining space
    # This prevents overfilling bins unnecessarily
    best_fit_component = -feasible_post_caps  # Higher score for less remaining space
    
    # Simplified diversity preservation component - calculate once instead of per bin
    # Calculate entropy of the post-placement state for the feasible bins only
    diversity_preservation_score = np.zeros_like(feasible_post_caps, dtype=float)
    
    # Calculate a simple diversity metric based on the variance of remaining capacities
    if len(feasible_post_caps) > 1:
        mean_cap = np.mean(feasible_post_caps)
        if mean_cap > 0:
            # Use variance as a proxy for diversity
            var_cap = np.var(feasible_post_caps)
            diversity_bonus = var_cap / (mean_cap + 1e-10)  # Normalize by mean to make scale-invariant
            
            # Give higher diversity scores to bins that preserve more diverse remaining capacities
            # This is approximated by giving each bin a score based on the general diversity
            diversity_preservation_score.fill(diversity_bonus)
    
    # Combine all three components with learned weights
    feasible_scores = (
        multiple_fit_weight * multiple_fit_score + 
        best_fit_weight * best_fit_component + 
        diversity_weight * diversity_preservation_score
    )
    
    # Assign the calculated scores to the correct positions
    scores[feasible_bins] = feasible_scores
    
    return scores
