def priority(item: float, bins_remain_cap: np.ndarray) -> np.ndarray:
    """
    Priority heuristic that integrates a lightweight online clustering module to identify
    latent capacity modes in feasible bins' post-placement remainders. Instead of using
    static multiplier embeddings, this approach applies fast density-based clustering
    (simulated with adaptive grouping) on feasible post-placement states to determine
    context-aware target remainders. This enables the multiple-fit score to prioritize
    placements that align with naturally emerging gap patterns while maintaining
    statelessness and capturing instance-specific structure more effectively.
    
    Implementation idea: Identify clusters of similar post-placement remainders among
    feasible bins and use cluster centroids as dynamic common sizes for the multiple-fit
    component. This replaces static multiplier embeddings with context-aware targets
    that emerge from the current bin state, potentially improving packing efficiency
    by targeting gaps that naturally form during the packing process.

    Args:
        item: Size of the item to place
        bins_remain_cap: NumPy array of remaining bin capacities
        
    Returns:
        NumPy array of priority scores for each bin
    """
    # Calculate post-placement remaining capacities
    post_placement_caps = bins_remain_cap - item
    
    # Initialize scores
    scores = np.full_like(bins_remain_cap, -np.inf, dtype=float)
    
    # Only consider bins that can accommodate the item
    feasible_bins = bins_remain_cap >= item
    
    if not np.any(feasible_bins):
        return scores  # All bins remain with -inf scores
    
    feasible_post_caps = post_placement_caps[feasible_bins]
    feasible_scores = np.zeros_like(feasible_post_caps, dtype=float)
    
    # Calculate the percentile rank of the current item among remaining bin capacities
    if len(bins_remain_cap) > 0:
        sorted_caps = np.sort(bins_remain_cap)
        item_percentile = np.searchsorted(sorted_caps, item) / len(sorted_caps)
    else:
        item_percentile = 0.5  # Default if no bins exist
    
    # Determine weights based on item size category
    if item_percentile > 0.82:  # Very large item
        best_fit_weight, multiple_fit_weight = 0.25 * 1.07, 0.75 * 0.99
    elif item_percentile < 0.18:  # Very small item
        best_fit_weight, multiple_fit_weight = 0.0005 * 0.93, 1.195 * 1.01
        best_fit_weight = max(best_fit_weight, 0.001)  # Ensure minimum best-fit influence
    elif item_percentile > 0.62:  # Large-medium item
        best_fit_weight, multiple_fit_weight = 0.14, 0.86
    elif item_percentile < 0.38:  # Small-medium item
        best_fit_weight, multiple_fit_weight = 0.025, 1.055
    else:  # True medium item
        best_fit_weight, multiple_fit_weight = 0.07, 0.93
    
    # Online clustering to find natural capacity modes in feasible post-placement remainders
    # We'll simulate density-based clustering by identifying local density peaks
    if len(feasible_post_caps) > 1:
        # Sort the post-placement capacities to find nearby values
        sorted_idx = np.argsort(feasible_post_caps)
        sorted_caps = feasible_post_caps[sorted_idx]
        
        # Define adaptive epsilon based on item size and range of feasible capacities
        cap_range = np.max(sorted_caps) - np.min(sorted_caps) if len(sorted_caps) > 1 else 0
        adaptive_eps = max(0.01, item * 0.1, cap_range * 0.05)
        
        # Find clusters by grouping nearby values
        cluster_centers = []
        i = 0
        while i < len(sorted_caps):
            center_start = sorted_caps[i]
            j = i
            # Group consecutive points within epsilon distance
            while j < len(sorted_caps) - 1 and (sorted_caps[j+1] - sorted_caps[j]) <= adaptive_eps:
                j += 1
            
            # Calculate the mean of the cluster as the center
            cluster_center = np.mean(sorted_caps[i:j+1])
            cluster_centers.append(cluster_center)
            
            i = j + 1
        
        # If no clusters formed, add the mean of all feasible caps
        if not cluster_centers:
            cluster_centers = [np.mean(feasible_post_caps)]
    else:
        # If only one feasible bin, use its post-capacity as the cluster center
        cluster_centers = [feasible_post_caps[0]] if len(feasible_post_caps) > 0 else []
    
    # Calculate multiple-fit score based on proximity to cluster centers
    multiple_fit_score = np.zeros_like(feasible_post_caps, dtype=float)
    
    # Add original item as a reference point
    all_targets = cluster_centers + [item]
    
    for target in all_targets:
        if target > 1e-9:  # Avoid division by very small numbers
            # Calculate how many multiples of target fit in the remaining capacity
            max_multiplier = int(np.max(feasible_post_caps) // target) + 2
            if max_multiplier > 0:
                # Calculate distances to all possible multiples
                distances_to_multiples = np.min([
                    np.abs(feasible_post_caps - n * target) 
                    for n in range(0, max_multiplier)
                ], axis=0)
                
                # Add to the score (higher score for closer matches)
                multiple_fit_score += 1.0 / (1.0 + distances_to_multiples)
    
    # Add Best Fit component: prefer bins with less remaining space
    best_fit_component = -feasible_post_caps  # Higher score for less remaining space
    
    # Combine all components with adaptive weights (without penalty term)
    feasible_scores = (
        multiple_fit_weight * multiple_fit_score + 
        best_fit_weight * best_fit_component
    )
    
    # Assign the calculated scores to the correct positions
    scores[feasible_bins] = feasible_scores
    
    return scores
