import numpy as np
from typing import List

def priority(item: float, bins_remain_cap: np.ndarray) -> np.ndarray:
    """
    Priority heuristic for online bin packing using a combination of best-fit and 
    multiple-fit strategies with simple lookahead estimation.
    
    For each feasible bin, calculates a priority score based on:
    1. How well the bin fits the current item (best-fit component)
    2. How well the remaining capacity aligns with common item sizes (multiple-fit component)
    3. A simple fragmentation penalty based on remaining capacity ratios

    Args:
        item: Size of the item to place
        bins_remain_cap: NumPy array of remaining bin capacities
        
    Returns:
        NumPy array of priority scores for each bin
    """
    # Initialize scores with negative infinity for invalid bins
    scores = np.full_like(bins_remain_cap, -np.inf, dtype=float)
    
    # Identify feasible bins (those with enough capacity for the item)
    feasible_bins = bins_remain_cap >= item
    
    if not np.any(feasible_bins):
        return scores  # Return all -inf if no feasible bins exist
    
    # Get remaining capacities for feasible bins only
    feasible_caps = bins_remain_cap[feasible_bins]
    post_placement_caps = feasible_caps - item  # After placing the current item
    
    # Calculate item percentile among all non-zero bin capacities for adaptive weighting
    non_zero_caps = bins_remain_cap[bins_remain_cap > 1e-9]
    if len(non_zero_caps) > 0:
        # Sort the non-zero capacities to find where the current item ranks
        sorted_caps = np.sort(non_zero_caps)
        # Calculate the percentile position of the current item
        item_percentile = np.searchsorted(sorted_caps, item) / len(sorted_caps)
    else:
        item_percentile = 0.5  # Default if no bins have capacity
    
    # Determine weights based on item size relative to existing capacities
    if item_percentile > 0.82:  # Very large item
        best_fit_weight, multiple_fit_weight = 0.25 * 1.07, 0.75 * 0.99
    elif item_percentile < 0.18:  # Very small item
        best_fit_weight, multiple_fit_weight = 0.001 * 0.93, 1.194 * 1.01
    elif item_percentile > 0.62:  # Large-medium item
        best_fit_weight, multiple_fit_weight = 0.14, 0.86
    elif item_percentile < 0.38:  # Small-medium item
        best_fit_weight, multiple_fit_weight = 0.025, 1.055
    else:  # True medium item
        best_fit_weight, multiple_fit_weight = 0.08, 0.92
    
    # Best-fit component: prefer bins that leave minimal remaining space
    best_fit_score = -post_placement_caps  # Negative because smaller remaining capacity is better
    
    # Multiple-fit component: prefer bins whose remaining capacity aligns well with common sizes
    multiple_fit_score = np.zeros_like(post_placement_caps, dtype=float)
    
    # Define common sizes to check against (current item and related sizes)
    common_sizes = np.array([item, item * 0.5, item * 1.5, item * 0.75, item * 1.25])
    
    # Add quantiles of current non-zero capacities as additional common sizes
    if len(non_zero_caps) > 0:
        remainder_quantiles = np.quantile(non_zero_caps, [0.25, 0.5, 0.75])
        common_sizes = np.concatenate([common_sizes, remainder_quantiles])
    
    # For each common size, calculate how well each bin's remaining capacity matches
    for common_size in common_sizes:
        if common_size > 1e-9:
            # Calculate the modulo to see how close the remaining capacity is to being a multiple
            mod_values = post_placement_caps % common_size
            # Prefer when the remainder is close to 0 (good fit) or close to the full common_size (leaves space for another)
            distance_to_good_fit = np.minimum(mod_values, common_size - mod_values)
            multiple_fit_score += 1.0 / (1.0 + distance_to_good_fit)
    
    # Combine the components to get the base priority score
    base_priority = (
        multiple_fit_weight * multiple_fit_score + 
        best_fit_weight * best_fit_score
    )
    
    # Simple lookahead: penalize bins that would create highly fragmented states
    # Fragmentation penalty based on ratio of remaining capacity to original bin capacity
    # (Assuming original capacity is approximately the max of current capacities)
    original_capacity_estimate = np.max(bins_remain_cap) if len(bins_remain_cap) > 0 else 1.0
    fragmentation_penalty = np.zeros_like(post_placement_caps)
    
    # If remaining capacity is a small fraction of original, it may lead to waste
    if original_capacity_estimate > 1e-9:
        capacity_ratio = post_placement_caps / original_capacity_estimate
        # Penalize very low remaining capacities (fragmentation) but not too heavily
        fragmentation_penalty = -0.1 * np.exp(-10 * capacity_ratio)  # Gentle penalty for small remainders
    
    # Final score combines base priority with fragmentation consideration
    final_scores = base_priority + fragmentation_penalty
    
    # Assign calculated scores to the corresponding positions in the full array
    scores[feasible_bins] = final_scores
    
    return scores

def priority(item: float, bins_remain_cap: np.ndarray) -> np.ndarray:

            mod_values = post_placement_caps % common_size
            # Prefer when the remainder is close to 0 (good fit) or close to the full common_size (leaves space for another)
            distance_to_good_fit = np.minimum(mod_values, common_size - mod_values)
            multiple_fit_score += 1.0 / (1.0 + distance_to_good_fit)
    
    # Combine the components to get the base priority score
    base_priority = (
        multiple_fit_weight * multiple_fit_score + 
        best_fit_weight * best_fit_score
    )
    
    # Monte Carlo lookahead simulation: estimate future costs for each feasible bin
    lookahead_scores = np.zeros_like(post_placement_caps, dtype=float)
    
    # Number of synthetic future items to simulate for lookahead
    num_lookahead_items = 3
    # Sample synthetic items from a distribution based on current remainders
    if len(non_zero_caps) > 0:
        # Create a distribution of synthetic items based on current remainder profile
        # Using a simple approach: sample from the non-zero capacities as they represent typical item sizes
        for i in range(len(post_placement_caps)):
            # Simulate placing the current item in bin i
            temp_bins = np.copy(bins_remain_cap)
            temp_bins[feasible_bins][i] -= item  # Place current item in this bin
            
            # Run lookahead simulation
            total_future_bins_used = 0
            
            for sim_idx in range(num_lookahead_items):
                # Sample a synthetic item based on current remainder profile
                # Use KDE-inspired approach: sample from existing non-zero remainders with some noise
                if len(temp_bins[temp_bins > 0]) > 0:
                    # Sample from current non-zero bin capacities with some variation
                    synthetic_item = np.random.choice(temp_bins[temp_bins > 0]) * np.random.uniform(0.3, 0.8)
                    # Ensure synthetic item is positive and not too large
                    synthetic_item = max(min(synthetic_item, np.max(temp_bins)), 1e-6)
                else:
                    # Fallback to a reasonable value based on original capacity
                    original_capacity = np.max(bins_remain_cap) if len(bins_remain_cap) > 0 else 100.0
                    synthetic_item = original_capacity * np.random.uniform(0.1, 0.5)
                
                # Simulate placement of synthetic item using a simple proxy heuristic (best-fit)
                valid_bins_for_synthetic = temp_bins >= synthetic_item
                if np.any(valid_bins_for_synthetic):
                    # Best-fit proxy: select bin with minimum remaining capacity after placement
                    temp_remaining_after_placement = np.where(valid_bins_for_synthetic, 
                                                              temp_bins - synthetic_item, 
                                                              np.inf)
                    best_bin_idx = np.argmin(temp_remaining_after_placement)
                    temp_bins[best_bin_idx] -= synthetic_item
                    total_future_bins_used += 1
                else:
                    # Need a new bin for the synthetic item
                    total_future_bins_used += 1
            
            # The lookahead score is inversely related to future bin usage (fewer bins is better)
            # So we subtract this cost from our base priority
            lookahead_scores[i] = -total_future_bins_used  # Negative because more future bins = worse choice
    
    # Apply lookahead with a weight
    lookahead_weight = 0.1  # Adjust this weight to balance lookahead importance
    final_scores = base_priority + lookahead_weight * lookahead_scores
    
    # Assign calculated scores to the corresponding positions in the full array
    scores[feasible_bins] = final_scores
    
    return scores
