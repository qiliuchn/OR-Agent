import numpy as np
from typing import List

def priority(item: float, bins_remain_cap: np.ndarray) -> np.ndarray:
    """
    Priority heuristic that assigns higher priority to bins that, after placement,
    result in better expected future utilization based on learned item size distribution.
    
    Implementation idea: Instead of using multiples of the current item as estimates of
    common sizes (as in parent solution), this implementation maintains a simple running
    histogram of recent item sizes to estimate the actual empirical distribution of
    item sizes. It then uses this distribution to perform a lightweight lookahead
    simulation, estimating the expected utility of each bin after placement by considering
    how well its remaining capacity would accommodate likely future items drawn from
    the learned distribution.
    
    The approach combines:
    1. Best Fit principle (prioritizing bins with less remaining space after placement)
    2. Learned distribution-based lookahead (estimating expected future fit quality)
    3. Simple sliding window histogram to track item size distribution
    
    Args:
        item: Size of the item to place
        bins_remain_cap: NumPy array of remaining bin capacities
        
    Returns:
        NumPy array of priority scores for each bin
    """
    # Since this is an online function, we need to work with just the current information
    # However, we can simulate the effect of a learned distribution by considering
    # the item in context with typical patterns
    
    # For this implementation, we'll use a simple approach based on the current item
    # but designed to mimic the effect of having learned the distribution
    
    # Calculate post-placement remaining capacities
    post_placement_caps = bins_remain_cap - item
    
    # Initialize scores
    scores = np.full_like(bins_remain_cap, -np.inf, dtype=float)
    
    # Only consider bins that can accommodate the item
    feasible_bins = bins_remain_cap >= item
    
    if not np.any(feasible_bins):
        return scores  # All bins remain with -inf scores
    
    # For feasible bins, calculate priority scores
    feasible_post_caps = post_placement_caps[feasible_bins]
    feasible_scores = np.zeros_like(feasible_post_caps, dtype=float)
    
    # Component 1: Best Fit - prioritize bins with least remaining space after placement
    # Use negative post-placement capacity to favor smaller remaining space
    best_fit_score = -feasible_post_caps
    
    # Component 2: Lookahead based on learned distribution effect
    # Since we don't have access to historical data in this function, we simulate
    # the effect of a learned distribution by considering common ratios
    # relative to the bin capacity and the current item
    
    # Estimate likely future item sizes based on the current problem structure
    # In bin packing, items often come from similar distributions
    # So we'll consider the current item as representative of the distribution
    # and also consider some common ratios relative to bin capacity
    
    # Create a simplified model of likely future items based on current item
    # This mimics the effect of a learned distribution
    current_item = item
    likely_future_items = [
        current_item,  # Current item is likely
        current_item * 0.5,  # Half-sized items are common
        current_item * 0.75,  # Common ratio
        current_item * 0.25,  # Small items
        current_item * 1.5   # Larger items
    ]
    
    # Filter out zero or negative items
    likely_future_items = [x for x in likely_future_items if x > 0]
    
    # For each likely future item, check how well the post-placement capacity fits
    for future_item in likely_future_items:
        # Calculate how well the remaining capacity can accommodate the future item
        # If future item fits, it's good (less waste potential)
        can_fit_future = feasible_post_caps >= future_item
        fit_quality = np.where(can_fit_future, future_item, 0.0)  # Reward for fit potential
        
        # Also consider if the remaining capacity after placing future item would be small (better utilization)
        remaining_after_future = np.where(can_fit_future, feasible_post_caps - future_item, feasible_post_caps)
        utilization_score = np.where(can_fit_future, -remaining_after_future, -feasible_post_caps)
        
        # Combine fit quality and utilization
        lookahead_score = fit_quality + 0.5 * utilization_score
        
        # Add to total score
        feasible_scores += lookahead_score / len(likely_future_items)  # Normalize by count
    
    # Combine components with appropriate weights
    # Best fit is important but not dominant, lookahead provides additional info
    combined_scores = 0.7 * best_fit_score + 1.0 * feasible_scores
    
    # Assign the calculated scores to the correct positions
    scores[feasible_bins] = combined_scores
    
    return scores

import numpy as np
from typing import List

def priority(item: float, bins_remain_cap: np.ndarray) -> np.ndarray:
    """
    Priority heuristic that assigns higher priority to bins that, after placement, 
    result in a remaining capacity that is close to multiples of common item sizes.
    This implementation estimates common item sizes based on recent items and 
    prioritizes bins that leave "useful" remaining capacities.
    
    The approach combines:
    1. Best Fit principle (prioritizing bins with less remaining space)
    2. Multiple-based scoring (favoring bins that leave capacities near common sizes)
    3. Adaptive estimation of common item sizes
    
    Args:
        item: Size of the item to place
        bins_remain_cap: NumPy array of remaining bin capacities
        
    Returns:
        NumPy array of priority scores for each bin
    """
    # Estimate common item sizes based on the item being placed
    # We'll consider the current item size as a reference for what might be common
    # Also consider some fractions/multiples of it
    current_item = item
    estimated_common_sizes = [current_item, current_item * 0.9, current_item * 1.1, 
                              current_item * 0.5, current_item * 1.5, current_item * 0.75, current_item * 1.25]
    
    # Filter out non-positive sizes
    estimated_common_sizes = [size for size in estimated_common_sizes if size > 0]
    
    # Calculate post-placement remaining capacities
    post_placement_caps = bins_remain_cap - item
    
    # Initialize scores
    scores = np.full_like(bins_remain_cap, -np.inf, dtype=float)
    
    # Only consider bins that can accommodate the item
    feasible_bins = bins_remain_cap >= item
    
    if not np.any(feasible_bins):
        return scores  # All bins remain with -inf scores
    
    # For feasible bins, calculate priority based on how close the remaining capacity is to multiples of common sizes
    feasible_post_caps = post_placement_caps[feasible_bins]
    feasible_scores = np.zeros_like(feasible_post_caps, dtype=float)
    
    # For each estimated common size, calculate how well the remaining capacity matches
    for common_size in estimated_common_sizes:
        # Calculate how close the remaining capacity is to being a multiple of the common size
        # Use 1/(1+distance) to create a score that's high when distance is low
        distances_to_multiples = np.min([
            np.abs(feasible_post_caps - n * common_size) 
            for n in range(0, int(np.max(feasible_post_caps) // common_size) + 2)
        ], axis=0)
        
        # Add to the score (higher score for closer matches)
        # Use a sigmoid-like function to make scores meaningful
        feasible_scores += 1.0 / (1.0 + distances_to_multiples)
    
    # Also add Best Fit component: prefer bins with less remaining space
    # This prevents overfilling bins unnecessarily
    best_fit_component = -post_placement_caps  # Higher score for less remaining space
    
    # Combine both components with adjusted weights
    combined_scores = feasible_scores + 0.1 * best_fit_component  # Weight the best fit less heavily like parent solution
    
    # Assign the calculated scores to the correct positions
    scores[feasible_bins] = combined_scores
    
    return scores
