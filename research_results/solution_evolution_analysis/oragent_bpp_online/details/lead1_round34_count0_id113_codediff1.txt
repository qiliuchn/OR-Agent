import numpy as np
from typing import List

def priority(item: float, bins_remain_cap: np.ndarray) -> np.ndarray:
    """
    Priority heuristic implementing an entropy-based meta-classifier that dynamically
    selects between specialized sub-heuristics based on real-time features of the
    current bin remainder distribution and item characteristics. This approach
    addresses the persistent underperformance on certain instances (like val_0/val_1)
    by detecting distributional patterns in the bin remainders and switching to
    appropriate specialized strategies without maintaining cross-instance state.
    
    The classifier uses lightweight statistics (entropy of remainders, coefficient of
    variation, and gap ratios) computed solely from the current state to determine
    whether the current packing scenario resembles clustered small items, sparse large
    items, or mixed patterns. Based on this classification, different combinations
    of best-fit and multiple-fit components are applied with tuned weights.
    
    This builds on successful elements from parent solutions including percentile-based
    item categorization and dynamic common-size generation, while introducing
    meta-adaptation based on distributional characteristics detectable at runtime.

    Args:
        item: Size of the item to place
        bins_remain_cap: NumPy array of remaining bin capacities
        
    Returns:
        NumPy array of priority scores for each bin
    """
    # Calculate post-placement remaining capacities
    post_placement_caps = bins_remain_cap - item
    
    # Initialize scores
    scores = np.full_like(bins_remain_cap, -np.inf, dtype=float)
    
    # Only consider bins that can accommodate the item
    feasible_bins = bins_remain_cap >= item
    
    if not np.any(feasible_bins):
        return scores  # All bins remain with -inf scores
    
    feasible_post_caps = post_placement_caps[feasible_bins]
    feasible_scores = np.zeros_like(feasible_post_caps, dtype=float)
    
    # Compute real-time features for meta-classification
    if len(bins_remain_cap[bins_remain_cap > 0]) > 1:
        # Entropy of normalized remainders
        positive_caps = bins_remain_cap[bins_remain_cap > 0]
        norm_caps = positive_caps / np.sum(positive_caps)  # Normalize to get probability-like distribution
        entropy = -np.sum(norm_caps * np.log(norm_caps + 1e-9))  # Add small value to avoid log(0)
        
        # Coefficient of variation
        cv = np.std(positive_caps) / (np.mean(positive_caps) + 1e-9)
        
        # Gap ratio between top quantiles
        q75 = np.percentile(positive_caps, 75)
        q25 = np.percentile(positive_caps, 25)
        gap_ratio = (q75 - q25) / (np.mean(positive_caps) + 1e-9) if len(positive_caps) > 1 else 1.0
        
        # Combined meta-feature
        meta_feature = entropy * cv * gap_ratio
    else:
        # Default values if insufficient data
        entropy = 1.0
        cv = 1.0
        gap_ratio = 1.0
        meta_feature = 1.0
    
    # Classify the current state based on the meta-feature
    # Low entropy + low CV = clustered scenario
    # High entropy + high CV = dispersed scenario
    # Mixed patterns based on gap ratio
    
    # Calculate the percentile rank of the current item among remaining bin capacities
    if len(bins_remain_cap) > 0:
        sorted_caps = np.sort(bins_remain_cap[bins_remain_cap > 0])
        if len(sorted_caps) > 0:
            item_percentile = np.searchsorted(sorted_caps, item) / len(sorted_caps)
        else:
            item_percentile = 0.5  # Default if no bins exist
    else:
        item_percentile = 0.5
    
    # Determine the meta-class and adjust strategy accordingly
    if entropy < 0.5 and cv < 0.8:
        # Clustered scenario - likely similar remaining capacities
        # Emphasize best-fit to maintain tight packing
        if item_percentile > 0.75:
            best_fit_weight, multiple_fit_weight = 0.3, 0.7
        elif item_percentile < 0.25:
            best_fit_weight, multiple_fit_weight = 0.05, 0.95
        else:
            best_fit_weight, multiple_fit_weight = 0.15, 0.85
    elif entropy > 1.2 or cv > 1.2:
        # Dispersed scenario - varied remaining capacities
        # Emphasize multiple-fit to find common sizes
        if item_percentile > 0.75:
            best_fit_weight, multiple_fit_weight = 0.1, 0.9
        elif item_percentile < 0.25:
            best_fit_weight, multiple_fit_weight = 0.02, 0.98
        else:
            best_fit_weight, multiple_fit_weight = 0.08, 0.92
    else:
        # Balanced scenario - moderate dispersion
        if item_percentile > 0.82:  # Very large item
            best_fit_weight, multiple_fit_weight = 0.25 * 1.07, 0.75 * 0.99
        elif item_percentile < 0.18:  # Very small item
            best_fit_weight, multiple_fit_weight = 0.001 * 0.93, 1.194 * 1.01
        elif item_percentile > 0.62:  # Large-medium item
            best_fit_weight, multiple_fit_weight = 0.14, 0.86
        elif item_percentile < 0.38:  # Small-medium item
            best_fit_weight, multiple_fit_weight = 0.025, 1.055
        else:  # True medium item
            best_fit_weight, multiple_fit_weight = 0.08, 0.92
    
    # Generate dynamic common sizes using context-aware sampling
    if len(feasible_post_caps) > 0:
        # Identify important scales from the current distribution
        unique_caps = np.unique(feasible_post_caps[feasible_post_caps > 1e-9])
        if len(unique_caps) > 0:
            # Sample key values from the current distribution as common sizes
            quantiles = np.array([0.1, 0.25, 0.33, 0.5, 0.67, 0.75, 0.9])
            quantile_values = np.quantile(unique_caps, quantiles)
            
            # Also include some ratios of the current item size
            item_ratios = np.array([item, item * 0.5, item * 1.5, item * 0.75, item * 1.25])
            
            # Include some simple fractions of the current remainders
            remainder_fractions = np.concatenate([
                unique_caps * 0.5,
                unique_caps * 0.33,
                unique_caps * 0.67,
                unique_caps * 0.25,
                unique_caps * 0.75
            ])
            
            # Combine all dynamically generated common sizes
            dynamic_common_sizes = np.concatenate([
                quantile_values,
                item_ratios,
                remainder_fractions
            ])
            
            # Remove duplicates and very small values
            dynamic_common_sizes = np.unique(dynamic_common_sizes)
            dynamic_common_sizes = dynamic_common_sizes[dynamic_common_sizes > 1e-9]
        else:
            # Fallback if no unique values exist
            dynamic_common_sizes = np.array([item, item * 0.5, item * 1.5])
    else:
        dynamic_common_sizes = np.array([item, item * 0.5, item * 1.5])
    
    # Limit the number of common sizes to prevent excessive computation
    if len(dynamic_common_sizes) > 50:
        # Select a representative subset
        step = len(dynamic_common_sizes) // 50
        dynamic_common_sizes = dynamic_common_sizes[::step][:50]
    
    # Calculate multiple fit score based on dynamic common sizes
    multiple_fit_score = np.zeros_like(feasible_post_caps, dtype=float)
    
    for common_size in dynamic_common_sizes:
        if common_size > 1e-9:  # Avoid division by very small numbers
            # Calculate distance to nearest multiple of the common size
            # Consider multiples up to a reasonable limit
            max_multiplier = min(int(np.max(feasible_post_caps) // common_size) + 2, 10)
            if max_multiplier > 0:
                # Calculate distances to all possible multiples
                distances_to_multiples = np.min([
                    np.abs(feasible_post_caps - n * common_size) 
                    for n in range(0, max_multiplier)
                ], axis=0)
                
                # Add to the score (higher score for closer matches)
                multiple_fit_score += 1.0 / (1.0 + distances_to_multiples)
    
    # Add Best Fit component: prefer bins with less remaining space
    best_fit_component = -feasible_post_caps  # Higher score for less remaining space
    
    # Combine all components with meta-class-specific weights
    feasible_scores = (
        multiple_fit_weight * multiple_fit_score + 
        best_fit_weight * best_fit_component
    )
    
    # Assign the calculated scores to the correct positions
    scores[feasible_bins] = feasible_scores
    
    return scores
