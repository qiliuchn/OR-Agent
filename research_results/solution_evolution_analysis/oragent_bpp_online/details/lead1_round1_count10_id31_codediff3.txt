            max_possible_entropy = np.log(len(all_positive_caps)) if len(all_positive_caps) > 1 else 1e-10

    adjusted_w_bestfit = max(0.1, base_w_bestfit - entropy_adjustment * 0.5)

    adjusted_w_diversity = min(0.6, base_w_diversity + entropy_adjustment * 0.2)

    total_weight = adjusted_w_bestfit + adjusted_w_multiple + adjusted_w_diversity
