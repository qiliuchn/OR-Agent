import numpy as np
from typing import List

def priority(item: float, bins_remain_cap: np.ndarray) -> np.ndarray:
    """
    Priority heuristic that adaptively selects different strategies based on item size regimes.
    
    Implementation idea: Replace the fixed linear combination of multiple-fit and best-fit components 
    with a non-linear, threshold-based switching mechanism that activates different heuristics 
    based on item size regimes:
    (a) for items >0.5*bin_cap, use pure Best Fit to avoid fragmentation;
    (b) for items between 0.2–0.5*bin_cap, use a balanced blend favoring multiple-fit scoring 
        toward high-frequency inferred sizes;
    (c) for items <0.2*bin_cap, emphasize diversity-aware scoring that penalizes bins whose 
        post-placement capacity clusters near existing residuals (using a kernel density estimate 
        approximated via pairwise distance penalties).
    
    Args:
        item: Size of the item to place
        bins_remain_cap: NumPy array of remaining bin capacities
        
    Returns:
        NumPy array of priority scores for each bin
    """
    # Calculate post-placement remaining capacities
    post_placement_caps = bins_remain_cap - item
    
    # Initialize scores
    scores = np.full_like(bins_remain_cap, -np.inf, dtype=float)
    
    # Only consider bins that can accommodate the item
    feasible_bins = bins_remain_cap >= item
    
    if not np.any(feasible_bins):
        return scores  # All bins remain with -inf scores
    
    feasible_post_caps = post_placement_caps[feasible_bins]
    feasible_scores = np.zeros_like(feasible_post_caps, dtype=float)
    
    # Determine bin capacity from the first element (all bins have same initial capacity)
    # We'll use a reference value to determine item size regime
    # Since the evaluation uses bins with capacity 100, we'll use that as a reference
    # Or we can infer from the max of bins_remain_cap as the initial capacity
    if len(bins_remain_cap) > 0:
        # Estimate bin capacity as maximum of initial bins (assuming they start full)
        bin_capacity = np.max(bins_remain_cap)  # This assumes bins start at full capacity
    else:
        bin_capacity = 100.0  # Default fallback
    
    # Define item size regimes
    large_item_threshold = 0.5 * bin_capacity
    medium_item_threshold = 0.2 * bin_capacity
    
    # Apply different heuristics based on item size regime
    if item > large_item_threshold:
        # Large item: Use pure Best Fit to avoid fragmentation
        feasible_scores = -feasible_post_caps  # Higher score for less remaining space after placement
    elif medium_item_threshold <= item <= large_item_threshold:
        # Medium item: Balanced blend favoring multiple-fit scoring
        # Calculate multiple-fit scoring
        estimated_common_sizes = [item, item * 0.9, item * 1.1, 
                                  item * 0.5, item * 1.5, item * 0.75, item * 1.25]
        estimated_common_sizes = [size for size in estimated_common_sizes if size > 0]
        
        multiple_fit_score = np.zeros_like(feasible_post_caps, dtype=float)
        
        for common_size in estimated_common_sizes:
            # Calculate distances to the nearest multiple of common_size
            multipliers = feasible_post_caps / common_size
            lower_mult = np.floor(multipliers)
            upper_mult = lower_mult + 1
            
            dist_to_lower = np.abs(feasible_post_caps - lower_mult * common_size)
            dist_to_upper = np.abs(feasible_post_caps - upper_mult * common_size)
            
            min_distances = np.minimum(dist_to_lower, dist_to_upper)
            
            # Add to the score (higher score for closer matches)
            multiple_fit_score += 1.0 / (1.0 + min_distances)
        
        # Best Fit component
        best_fit_component = -feasible_post_caps
        
        # Combine with emphasis on multiple-fit scoring
        feasible_scores = multiple_fit_score + 0.3 * best_fit_component
    else:
        # Small item: Diversity-aware scoring that penalizes clustering
        # Calculate multiple-fit scoring first
        estimated_common_sizes = [item, item * 0.9, item * 1.1, 
                                  item * 0.5, item * 1.5, item * 0.75, item * 1.25]
        estimated_common_sizes = [size for size in estimated_common_sizes if size > 0]
        
        multiple_fit_score = np.zeros_like(feasible_post_caps, dtype=float)
        
        for common_size in estimated_common_sizes:
            # Calculate distances to the nearest multiple of common_size
            multipliers = feasible_post_caps / common_size
            lower_mult = np.floor(multipliers)
            upper_mult = lower_mult + 1
            
            dist_to_lower = np.abs(feasible_post_caps - lower_mult * common_size)
            dist_to_upper = np.abs(feasible_post_caps - upper_mult * common_size)
            
            min_distances = np.minimum(dist_to_lower, dist_to_upper)
            
            # Add to the score (higher score for closer matches)
            multiple_fit_score += 1.0 / (1.0 + min_distances)
        
        # Diversity-aware penalty: penalize bins whose post-placement capacity is similar to others
        diversity_penalty = np.zeros_like(feasible_post_caps, dtype=float)
        
        if len(feasible_post_caps) > 1:
            # For each bin, calculate how close its post-capacity is to other bins' post-capacities
            for i in range(len(feasible_post_caps)):
                distances_to_others = np.abs(feasible_post_caps - feasible_post_caps[i])
                # Ignore self-distance (which would be 0)
                distances_to_others = np.where(np.arange(len(feasible_post_caps)) == i, np.inf, distances_to_others)
                # Sum up inverse of distances to measure clustering (smaller sum means more diverse)
                # Use a small epsilon to avoid division by zero
                inv_distances = np.where(distances_to_others == np.inf, 0, 1.0 / (1e-6 + distances_to_others))
                diversity_penalty[i] = -np.sum(inv_distances)  # Negative because we want to penalize clustering
        
        # Combine components
        feasible_scores = multiple_fit_score + diversity_penalty
    
    # Assign the calculated scores to the correct positions
    scores[feasible_bins] = feasible_scores
    
    return scores

import numpy as np

def priority(item: float, bins_remain_cap: np.ndarray) -> np.ndarray:
    """
    Priority heuristic that assigns higher priority to bins that, after placement, 
    result in a remaining capacity that is close to multiples of common item sizes.
    This implementation estimates common item sizes based on recent items and 
    prioritizes bins that leave "useful" remaining capacities.
    
    The approach combines:
    1. Best Fit principle (prioritizing bins with less remaining space)
    2. Multiple-based scoring (favoring bins that leave capacities near common sizes)
    
    Args:
        item: Size of the item to place
        bins_remain_cap: NumPy array of remaining bin capacities
        
    Returns:
        NumPy array of priority scores for each bin
    """
    # Calculate post-placement remaining capacities
    post_placement_caps = bins_remain_cap - item
    
    # Initialize scores
    scores = np.full_like(bins_remain_cap, -np.inf, dtype=float)
    
    # Only consider bins that can accommodate the item
    feasible_bins = bins_remain_cap >= item
    
    if not np.any(feasible_bins):
        return scores  # All bins remain with -inf scores
    
    feasible_post_caps = post_placement_caps[feasible_bins]
    feasible_scores = np.zeros_like(feasible_post_caps, dtype=float)
    
    # Estimate common item sizes based on the item being placed
    current_item = item
    estimated_common_sizes = [current_item, current_item * 0.9, current_item * 1.1, 
                              current_item * 0.5, current_item * 1.5, current_item * 0.75, current_item * 1.25]
    
    # Filter out non-positive sizes
    estimated_common_sizes = [size for size in estimated_common_sizes if size > 0]
    
    # Calculate how close the remaining capacity is to being a multiple of common sizes
    multiple_fit_score = np.zeros_like(feasible_post_caps, dtype=float)
    
    for common_size in estimated_common_sizes:
        # Calculate distances to the nearest multiple of common_size
        multipliers = feasible_post_caps / common_size
        lower_mult = np.floor(multipliers)
        upper_mult = lower_mult + 1
        
        dist_to_lower = np.abs(feasible_post_caps - lower_mult * common_size)
        dist_to_upper = np.abs(feasible_post_caps - upper_mult * common_size)
        
        min_distances = np.minimum(dist_to_lower, dist_to_upper)
        
        # Add to the score (higher score for closer matches)
        multiple_fit_score += 1.0 / (1.0 + min_distances)
    
    # Add Best Fit component: prefer bins with less remaining space
    # This prevents overfilling bins unnecessarily
    best_fit_component = -feasible_post_caps  # Higher score for less remaining space
    
    # Combine both components
    # Use smaller weight for best fit component to balance between multiple-based scoring and best fit
    feasible_scores = multiple_fit_score + 0.1 * best_fit_component
    
    # Assign the calculated scores to the correct positions
    scores[feasible_bins] = feasible_scores
    
    return scores
