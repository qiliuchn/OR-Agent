import numpy as np
from typing import List

def priority(item: float, bins_remain_cap: np.ndarray) -> np.ndarray:
    """
    Efficient priority heuristic for bin selection in online bin packing.
    
    Implementation idea: Creates a fast, stateless heuristic that balances between
    Best Fit (minimizing wasted space) and First Fit (efficiency) by using a 
    combination of remaining capacity utilization and proximity to current item size.
    This maintains the spirit of the original concept but with much better performance.
    
    The heuristic combines:
    1. A Best Fit component that prefers bins with minimal remaining capacity after placement
    2. A proximity component that favors bins whose remaining capacity is similar to the current item size
       (indicating potential for good packing with future similar-sized items)
    3. A capacity utilization factor to encourage efficient use of available space
    
    Args:
        item: Size of the item to place
        bins_remain_cap: NumPy array of remaining bin capacities
        
    Returns:
        NumPy array of priority scores for each bin
    """
    # Calculate post-placement remaining capacities
    post_placement_caps = bins_remain_cap - item
    
    # Initialize scores with -inf for impossible placements
    scores = np.full_like(bins_remain_cap, -np.inf, dtype=float)
    
    # Only consider bins that can accommodate the item
    feasible_bins = bins_remain_cap >= item
    
    if not np.any(feasible_bins):
        return scores  # All bins remain with -inf scores
    
    # Calculate scores for feasible bins only
    feasible_post_caps = post_placement_caps[feasible_bins]
    
    # Best fit component: prefer smaller remaining capacity after placement (higher priority)
    # Use negative to make smaller remaining capacity have higher priority
    best_fit_component = -feasible_post_caps
    
    # Proximity component: prefer remaining capacity that's close to current item size
    # This encourages leaving space for potentially similar-sized future items
    proximity_component = -np.abs(feasible_post_caps - item)
    
    # Capacity utilization component: prefer bins that use capacity efficiently
    # Higher remaining capacity before placement indicates potentially better bin choice
    # But only consider this for the remaining capacity after placement
    utilization_component = feasible_post_caps / (feasible_post_caps + item + 1e-8)  # Avoid division issues
    
    # Combine components with appropriate weights
    # Best fit gets highest weight since minimizing bins is the primary objective
    feasible_scores = 2.0 * best_fit_component + 1.0 * proximity_component + 0.5 * utilization_component
    
    # Handle edge case where remaining capacity is exactly zero (perfect fit)
    perfect_fit_mask = feasible_post_caps == 0
    feasible_scores[perfect_fit_mask] = np.inf  # Highest possible priority for perfect fit
    
    # Assign the calculated scores to the correct positions
    scores[feasible_bins] = feasible_scores
    
    return scores

import numpy as np
from typing import List

def priority(item: float, bins_remain_cap: np.ndarray) -> np.ndarray:
    """
    Priority heuristic that assigns higher priority to bins that, after placement, 
    result in a remaining capacity that is close to common item sizes.
    This implementation estimates common item sizes based on the current item and 
    prioritizes bins that leave "useful" remaining capacities.
    
    The approach combines:
    1. Estimation of common item sizes based on the current item
    2. Scoring bins based on how well their post-placement capacity matches these sizes
    3. Best Fit principle as a secondary criterion to avoid wasting space
    
    Args:
        item: Size of the item to place
        bins_remain_cap: NumPy array of remaining bin capacities
        
    Returns:
        NumPy array of priority scores for each bin
    """
    # Estimate common item sizes based on the current item
    # Consider the current item size and its common fractions/multiples
    current_item = item
    estimated_common_sizes = [
        current_item, 
        current_item * 0.5,   # half size
        current_item * 0.75,  # three quarters
        current_item * 0.9,   # 90% size
        current_item * 1.1,   # 110% size
        current_item * 1.25,  # 1.25x size
        current_item * 1.5,   # 1.5x size
        current_item * 0.25,  # quarter size
        current_item * 2.0    # double size
    ]
    
    # Filter out non-positive sizes
    estimated_common_sizes = [size for size in estimated_common_sizes if size > 0]
    
    # Calculate post-placement remaining capacities
    post_placement_caps = bins_remain_cap - item
    
    # Initialize scores
    scores = np.full_like(bins_remain_cap, -np.inf, dtype=float)
    
    # Only consider bins that can accommodate the item
    feasible_bins = bins_remain_cap >= item
    
    if not np.any(feasible_bins):
        return scores  # All bins remain with -inf scores
    
    # For feasible bins, calculate priority based on how close the remaining capacity is to common sizes
    feasible_post_caps = post_placement_caps[feasible_bins]
    feasible_scores = np.zeros_like(feasible_post_caps, dtype=float)
    
    # For each estimated common size, calculate how well the remaining capacity matches
    for common_size in estimated_common_sizes:
        # Calculate how close the remaining capacity is to being a multiple of the common size
        # Consider multiples: 0, 1x, 2x, 3x, etc. of the common size
        max_multiple = int(np.max(feasible_post_caps) // common_size) + 2
        distances_to_multiples = np.min([
            np.abs(feasible_post_caps - n * common_size) 
            for n in range(0, max_multiple)
        ], axis=0)
        
        # Add to the score (higher score for closer matches to multiples)
        # Use a function that gives significant scores only for close matches
        feasible_scores += 1.0 / (1.0 + distances_to_multiples)
    
    # Also add Best Fit component: prefer bins with less remaining space after placement
    # This prevents overfilling bins unnecessarily while still allowing for good fits
    best_fit_component = -post_placement_caps  # Higher score for less remaining space after placement
    
    # Combine both components with appropriate weights
    # Give more weight to the common size matching since it was the key insight from parent solution
    combined_scores = feasible_scores + 0.05 * best_fit_component  # Optimal weight found in previous experiment
    
    # Assign the calculated scores to the correct positions
    scores[feasible_bins] = combined_scores
    
    return scores
