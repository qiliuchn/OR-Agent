    # Use a more conservative approach: start with the proven 0.1 and make small adjustments
    base_penalty = 0.1
    entropy_factor = min(normalized_entropy * 0.3, 0.2)  # Limit impact of entropy
    frag_factor = min(current_fragmentation_index * 0.4, 0.2)  # Limit impact of current fragmentation
    adaptive_penalty_strength = base_penalty * (1.0 + entropy_factor - frag_factor * 0.5)  # Reduce penalty when current fragmentation is high
