import numpy as np
from collections import deque
from typing import Deque

def priority(item: float, bins_remain_cap: np.ndarray) -> np.ndarray:
    """
    Priority heuristic that maintains a dynamic estimate of common item sizes
    using a sliding window approach. The algorithm keeps track of recent item sizes
    in a histogram-like structure to better predict future packing opportunities.
    
    The approach combines:
    1. Dynamic common size estimation using a sliding window of recent items
    2. Best Fit principle (prioritizing bins with less remaining space after placement)
    3. Multiple-based scoring (favoring bins that leave capacities near common sizes)
    
    Implementation idea: Instead of estimating common sizes only from the current item,
    we maintain a history of recent item sizes to identify more accurate patterns in
    the input distribution. This allows the algorithm to adapt to changes in the
    item size distribution during the online packing process.
    
    Args:
        item: Size of the item to place
        bins_remain_cap: NumPy array of remaining bin capacities
        
    Returns:
        NumPy array of priority scores for each bin
    """
    # Static attributes to maintain state across function calls
    if not hasattr(priority, 'recent_items'):
        priority.recent_items: Deque[float] = deque(maxlen=50)  # Sliding window of last 50 items
        priority.common_sizes_cache = []  # Cached common sizes from recent items
    
    # Update the recent items buffer with the current item
    priority.recent_items.append(item)
    
    # Update common sizes cache based on recent items
    if len(priority.recent_items) > 0:
        # Calculate statistics about recent items
        recent_array = np.array(priority.recent_items)
        unique_sizes = np.unique(recent_array)
        
        # Generate common sizes based on recent observations
        estimated_common_sizes = []
        
        # Add actual recent unique sizes
        estimated_common_sizes.extend(unique_sizes.tolist())
        
        # Add common fractions and multiples of recent sizes
        for size in unique_sizes:
            if size > 0:
                estimated_common_sizes.extend([
                    size * 0.5,   # half
                    size * 0.75,  # three quarters
                    size * 1.25,  # 1.25 times
                    size * 1.5,   # 1.5 times
                    size * 2.0    # double
                ])
        
        # Remove duplicates and filter positive values
        estimated_common_sizes = list(set(size for size in estimated_common_sizes if size > 0))
        
        # Sort and limit the number of common sizes to avoid too much computation
        estimated_common_sizes.sort()
        priority.common_sizes_cache = estimated_common_sizes[:20]  # Limit to top 20 candidates
    else:
        # Fallback if no recent items
        priority.common_sizes_cache = [item, item * 0.5, item * 1.5, item * 2.0]
    
    # Calculate post-placement remaining capacities
    post_placement_caps = bins_remain_cap - item
    
    # Initialize scores
    scores = np.full_like(bins_remain_cap, -np.inf, dtype=float)
    
    # Only consider bins that can accommodate the item
    feasible_bins = bins_remain_cap >= item
    
    if not np.any(feasible_bins):
        return scores  # All bins remain with -inf scores
    
    # For feasible bins, calculate priority based on how close the remaining capacity is to multiples of common sizes
    feasible_post_caps = post_placement_caps[feasible_bins]
    feasible_scores = np.zeros_like(feasible_post_caps, dtype=float)
    
    # For each estimated common size, calculate how well the remaining capacity matches
    for common_size in priority.common_sizes_cache:
        # Calculate how close the remaining capacity is to being a multiple of the common size
        # Use max(1, ...) to avoid division by zero and ensure reasonable scores
        max_multiple = max(1, int(np.max(feasible_post_caps) // common_size) + 2)
        
        # Compute distances to nearest multiples
        distances_to_multiples = np.min([
            np.abs(feasible_post_caps - n * common_size) 
            for n in range(0, max_multiple)
        ], axis=0)
        
        # Add to the score (higher score for closer matches)
        # Use a sigmoid-like function to make scores meaningful
        feasible_scores += 1.0 / (1.0 + distances_to_multiples)
    
    # Also add Best Fit component: prefer bins with less remaining space
    # This prevents overfilling bins unnecessarily
    best_fit_component = -post_placement_caps  # Higher score for less remaining space
    
    # Combine both components
    combined_scores = feasible_scores + 0.1 * best_fit_component  # Weight the best fit less heavily
    
    # Assign the calculated scores to the correct positions
    scores[feasible_bins] = combined_scores
    
    return scores

def priority(item: float, bins_remain_cap: np.ndarray) -> np.ndarray:

    Args:

    post_placement_caps = bins_remain_cap - item

    feasible_bins = bins_remain_cap >= item
    
    if not np.any(feasible_bins):
        return scores  # All bins remain with -inf scores
    
    # For feasible bins, calculate priority based on how close the remaining capacity is to multiples of common sizes
    feasible_post_caps = post_placement_caps[feasible_bins]
    feasible_scores = np.zeros_like(feasible_post_caps, dtype=float)
    
    # Estimate common item sizes based on the current item
    # Consider the current item and some fractions/multiples of it
    current_item = item
    estimated_common_sizes = [
        current_item, 
        current_item * 0.5,   # half
        current_item * 0.75,  # three quarters
        current_item * 0.9,   # 90%

        current_item * 1.25,  # 1.25 times

    # Filter out non-positive sizes

    best_fit_component = -post_placement_caps  # Higher score for less remaining space

    combined_scores = feasible_scores + 0.2 * best_fit_component  # Increase weight of best fit component
